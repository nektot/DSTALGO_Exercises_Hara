// Number 2
Console.Write("Quiz: ");
int quiz = Convert.ToInt32(Console.ReadLine());

Console.Write($"Midterm: ");
int midterm = Convert.ToInt32(Console.ReadLine());

Console.Write($"Final: ");
int final = Convert.ToInt32(Console.ReadLine());

double finalGrade = (quiz * 0.2) + (midterm * 0.3) + (final * 0.5);
Console.WriteLine($"Your final grade is: {finalGrade}");

bool isPassed = finalGrade >= 75;
bool isExcellent = finalGrade >= 90;
bool isFailed = finalGrade < 75;

Console.WriteLine($"Passed: {isPassed}");
Console.WriteLine($"Excellent: {isExcellent}");
Console.WriteLine($"Failed: {isFailed}");
